## 一、数据模型层（2 个文件）

### 1. 新建 `StudyPulse/Models/ExamReversePlannerModels.swift`

定义 5 个 `nonisolated struct`，全部 `Identifiable, Codable, Hashable, Sendable`，全部带自定义 `init(from:)` + `decodeIfPresent` 兼容老数据（照搬 `ExamSimulationModels.swift` 范式）：

```swift
// 用户输入的目标（可独立 CRUD）
struct ExamGoal {
    var id: UUID
    var examName: String
    var subject: String
    var examDate: Date
    var currentScore: Double
    var targetScore: Double
    var fullScore: Double
    var phaseId: UUID?
    var createdAt: Date
}

// AI 产出的薄弱点（ExamPlan 的子结构）
struct WeakPoint {
    var id: UUID
    var topic: String
    var mastery: Double          // 0~1
    var possibleScoreGain: Double
    var priority: Int            // 1=最高
}

// 分阶段计划（Day 1-5 / Day 6-10 ...）
struct PlanPhase {
    var id: UUID
    var name: String
    var dayRange: String         // "1-5"
    var goal: String
}

// 每日任务（含 reason）
struct DailyExamTask {
    var id: UUID
    var dayOffset: Int           // Day n（相对今天）
    var date: Date               // 计算后的具体日期
    var subject: String
    var durationMinutes: Int
    var taskTitle: String
    var reason: String
}

// AI 生成的完整计划（聚合根）
struct ExamPlan {
    var id: UUID
    var examGoalID: UUID
    var improvementTarget: Double
    var summary: String          // 整体差距分析文本
    var weakPoints: [WeakPoint]
    var phases: [PlanPhase]
    var dailyTasks: [DailyExamTask]
    var modelInfo: String?       // 记录是哪个 LLM 生成的，便于 Debug
    var createdAt: Date
}
```

### 5. 新建 `StudyPulse/Repositories/Protocols/ExamPlanRepository.swift`

```swift
@MainActor
protocol ExamPlanRepository: AnyObject, Sendable {
    var goals: [ExamGoal] { get }
    var plans: [ExamPlan] { get }
    func loadAll(context: ModelContext) async
    func upsertGoal(_ goal: ExamGoal)
    func deleteGoal(_ goal: ExamGoal)           // 级联删除该 goal 的所有 plan
    func upsertPlan(_ plan: ExamPlan)
    func deletePlan(_ plan: ExamPlan)
}

extension ExamPlanRepository {
    func plans(for goalID: UUID) -> [ExamPlan] {
        plans.filter { $0.examGoalID == goalID }.sorted { $0.createdAt > $1.createdAt }
    }
    func latestPlan(for goalID: UUID) -> ExamPlan? { plans(for: goalID).first }
}
```

## 三、LLM Service 层（1 个文件）

### 8. 新建 `StudyPulse/Managers/LLM/ExamReversePlannerLLM.swift`

照搬 `ExamAutopsyLLM` 的 namespace enum 模式（这是项目里最新、最简洁的 JSON-输出 caller 模板）：

```swift
enum ExamReversePlannerLLM {
    static let caller = "ExamReversePlanner"

    private static let system = """
        你是一名专业学习规划专家。你的任务是根据学生当前学习状态，
        从目标考试成绩反推需要完成的提升路径。
        ... (用户文档里的 system prompt 原文)
        【输出格式强制规则】
        只返回合法 JSON，不要 markdown 代码块、不要解释文字。
        JSON schema: { summary, weakPoints:[...], phases:[...], dailyTasks:[...] }
        """

    @MainActor
    static func generate(goal: ExamGoal, container: RepositoryContainer) async throws -> ExamPlan {
        let config = container.envManager.llmConfig
        guard config.isConfigured else { throw LLMError.notConfigured }

        // 组装 4 类上下文（按用户确认的范围）
        let context = ContextBuilder.build(goal: goal, container: container)
        let prompt = LLMPrompt(system: system, messages: [.user(context)])
        let raw = try await LLMClient.shared.complete(
            prompt: prompt, config: config, caller: caller
        )
        return try parse(raw, goal: goal, modelInfo: config.model)
    }

    nonisolated
    static func parse(_ raw: String, goal: ExamGoal, modelInfo: String?) throws -> ExamPlan {
        // 1. 大括号扫描（firstIndex "{" ... lastIndex "}"），剥离 ```json``` 噪声
        // 2. JSONSerialization → 字典
        // 3. private struct RawPlan: Codable { 全部字段可选 + snapshot() 带默认值 }
        // 4. 把 dayOffset 转成具体 date（基于 goal.examDate 倒推 / 或基于今天正推）
        // 5. 返回 ExamPlan
    }
}
```

**`ContextBuilder`（嵌套在同一个 enum 里）** 负责 4 类上下文组装：
- `gradeRepo.grades.filter { $0.subject == goal.subject }.suffix(15)` → 近期成绩均值 + 趋势
- `mistakeRepo.mistakeSets.filter { $0.subject == goal.subject }` → 按 `tags` 聚合 top 5 薄弱 tag（用 `Dictionary(grouping:)` 手动聚合，**不用 uniqueKeysWithValues** 避免 AGENTS.md 第 16 节警告的重复键崩溃）
- 错题的 SRS 队列状态：overdue 数（`reviewState?.nextReviewDate < Date()`）+ 总待复习数
- `taskRepo` 该科目的未完成 Todo 数
- 全部 JSON-encode 后塞进 user message，**手动截断**（成绩最多 15 条、错题最多 30 条、tag top 5）控制 token 预算

**自动享受的能力**（无需手写）：
- BYOK / Cloud AI 透明路由（`LLMClient` 内部 `if config.isCloudProvider` 分发）
- LLMResponseCache 按 caller 命名空间缓存（10 分钟 TTL）
- DEBUG 模式 `LLMDebugSheet` 自动按 caller 分组显示

## 四、ViewModel 层（1 个文件）

### 9. 新建 `StudyPulse/ViewModels/ExamReversePlannerViewModel.swift`

照搬 `ExamSimulationViewModel` + `AutoMindMapViewModel` 的混合范式（`@MainActor @Observable final class` + `makeDefault` 工厂 + 状态机 + LLM 调用 + 本地降级）：

```swift
@MainActor @Observable
final class ExamReversePlannerViewModel {
    enum Phase: Equatable { case idle, loading, success, failure }

    private(set) var phase: Phase = .idle
    private(set) var goals: [ExamGoal] = []
    private(set) var selectedGoalID: UUID?
    private(set) var currentPlan: ExamPlan?
    private(set) var errorMessage: String?

    private let container: RepositoryContainer

    init(container: RepositoryContainer) { self.container = container; refreshGoals() }
    static func makeDefault(container: RepositoryContainer) -> ExamReversePlannerViewModel { .init(container: container) }

    var selectedGoal: ExamGoal? { goals.first { $0.id == selectedGoalID } }

    func refreshGoals() { goals = container.examPlanRepo.goals.sorted { $0.createdAt > $1.createdAt } }

    func saveGoal(_ goal: ExamGoal) {            // 创建 / 编辑
        container.examPlanRepo.upsertGoal(goal)
        refreshGoals()
        selectedGoalID = goal.id
    }

    func deleteGoal(_ goal: ExamGoal) {
        container.examPlanRepo.deleteGoal(goal)
        refreshGoals()
    }

    func loadLatestPlan(for goalID: UUID) { currentPlan = container.examPlanRepo.latestPlan(for: goalID) }

    func generatePlan(for goal: ExamGoal) async {  // 主流程
        phase = .loading; errorMessage = nil
        do {
            let plan = try await ExamReversePlannerLLM.generate(goal: goal, container: container)
            container.examPlanRepo.upsertPlan(plan)
            currentPlan = plan
            phase = .success
        } catch is CancellationError {
            phase = .idle
        } catch {
            let desc = (error as? LLMError)?.errorDescription ?? error.localizedDescription
            Log.llm.error("ExamReversePlanner 失败: \(desc, privacy: .public)")
            errorMessage = desc
            phase = .failure
        }
    }

    func deletePlan(_ plan: ExamPlan) {
        container.examPlanRepo.deletePlan(plan)
        if currentPlan?.id == plan.id { currentPlan = nil }
    }
}
```

## 八、不做什么（明确边界）

- **不**做需求文档"功能联动 3"的 Todo 标题自动匹配（扫 Todo 文本生成 Goal）—— 这种基于字符串匹配的关联不可靠，改由 ExamDetailView toolbar 入口实现"考试 → 倒推计划"的强关联（更稳）。
- **不**做需求文档"功能联动 4"的 AI Coach 主页注入 —— 等 ExamPlan 跑通后再做扩展。
- **不**写单元测试（项目惯例不强制，与 ExamSimulation 一致；如需可后补 `ExamReversePlannerLLMTests` 测 JSON 解析）。
- **不**改动 CoachView（按用户决策，挂 Home 卡片 + ExamDetailView）。
- **不**新增 facade 方法到 RepositoryContainer（与 examSimulationRepo 一致，View 直接调 `container.examPlanRepo.xxx`）。
